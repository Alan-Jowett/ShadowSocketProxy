44/// Lifecycle state for a Windows conditional-accept request.
45#[derive(Debug, Clone, Copy, PartialEq, Eq)]
46#[repr(u8)]
47#[cfg(any(target_os = "windows", test))]
48enum ConditionalAttemptState {
49    /// Request has been deferred and awaits preconnect.
50    Deferred = 0,
51    /// Worker owns a preconnected outbound socket.
52    Ready = 1,
53    /// Callback permitted `WSAAccept` to return the client socket.
54    Claimed = 2,
55    /// Request must receive conditional rejection.
56    Rejected = 3,
57    /// Shutdown cancelled the request before completion.
58    Cancelled = 4,
59}
60
61#[cfg(any(target_os = "windows", test))]
62impl ConditionalAttemptState {
63    /// Decodes the atomic representation, treating unknown values as cancelled.
64    fn from_raw(value: u8) -> Self {
65        match value {
66            0 => Self::Deferred,
67            1 => Self::Ready,
68            2 => Self::Claimed,
69            3 => Self::Rejected,
70            _ => Self::Cancelled,
71        }
72    }
73}

86/// State shared by the conditional callback, coordinator, and preconnect
87/// worker. It contains no socket ownership so it is also testable off Windows.
88#[cfg(any(target_os = "windows", test))]
89struct ConditionalAttempt {
90    /// Immutable identity for this deferred attempt.
91    id: ConditionalRequestId,
92    /// Timestamp of the initial `CF_DEFER`.
93    #[cfg(target_os = "windows")]
94    deferred_at: std::time::Instant,
95    /// Atomic lifecycle state shared across native contexts.
96    state: AtomicU8,
97    /// Ensures the admission count is decremented at most once.
98    released: AtomicBool,
99}
100
101#[cfg(any(target_os = "windows", test))]
102impl ConditionalAttempt {
103    /// Creates a request in the deferred state.
104    fn new(id: ConditionalRequestId) -> Self {
105        Self {
106            id,
107            #[cfg(target_os = "windows")]
108            deferred_at: std::time::Instant::now(),
109            state: AtomicU8::new(ConditionalAttemptState::Deferred as u8),
110            released: AtomicBool::new(false),
111        }
112    }
113
114    /// Reads the atomically published lifecycle state.
115    fn state(&self) -> ConditionalAttemptState {
116        ConditionalAttemptState::from_raw(self.state.load(Ordering::Acquire))
117    }
118}

120/// Atomically limits pending Windows conditional-accept attempts and gives
121/// every admitted request a monotonically increasing generation.
122#[cfg(any(target_os = "windows", test))]
123struct ConditionalAdmissions {
124    /// Maximum number of unaccepted deferred attempts.
125    limit: usize,
126    /// Currently reserved deferred attempts.
127    pending: AtomicUsize,
128    /// Source of unique request-generation values.
129    next_generation: AtomicU64,
130}
131
132#[cfg(any(target_os = "windows", test))]
133impl ConditionalAdmissions {
134    /// Creates an admission controller with the configured limit.
135    fn new(limit: usize) -> Self {
136        Self {
137            limit,
138            pending: AtomicUsize::new(0),
139            next_generation: AtomicU64::new(1),
140        }
141    }
142
143    /// Reserves one bounded attempt and assigns its request generation.
144    fn reserve(&self, tuple: Tuple) -> Option<ConditionalAttempt> {
145        let mut current = self.pending.load(Ordering::Acquire);
146        loop {
147            if current >= self.limit {
148                return None;
149            }
150            match self.pending.compare_exchange_weak(
151                current,
152                current + 1,
153                Ordering::AcqRel,
154                Ordering::Acquire,
155            ) {
156                Ok(_) => {
157                    let generation = self.next_generation.fetch_add(1, Ordering::Relaxed);
158                    return Some(ConditionalAttempt::new(ConditionalRequestId {
159                        tuple,
160                        generation,
161                    }));
162                }
163                Err(observed) => current = observed,
164            }
165        }
166    }
167
168    /// Releases a reservation once, including on stale completion and shutdown
169    /// paths that race the conditional callback.
170    fn release(&self, attempt: &ConditionalAttempt) {
171        if !attempt.released.swap(true, Ordering::AcqRel) {
172            self.pending.fetch_sub(1, Ordering::AcqRel);
173        }
174    }
175
176    #[cfg(test)]
177    fn pending(&self) -> usize {
178        self.pending.load(Ordering::Acquire)
179    }
180}

182/// Computes the remaining conditional-preconnect budget from the first defer,
183/// not from when a worker happens to start.
184#[cfg(any(target_os = "windows", test))]
185fn conditional_preconnect_budget(deferred_at: std::time::Instant) -> Result<Duration, ProxyError> {
186    CONTROL_RPC_TIMEOUT
187        .checked_sub(deferred_at.elapsed())
188        .ok_or_else(|| {
189            ProxyError::Control("conditional preconnect timed out before it started".into())
190        })
191}

193/// Changes a deferred request to ready only once its outbound socket is stored.
194#[cfg(any(target_os = "windows", test))]
195fn mark_attempt_ready(attempt: &ConditionalAttempt) -> bool {
196    attempt
197        .state
198        .compare_exchange(
199            ConditionalAttemptState::Deferred as u8,
200            ConditionalAttemptState::Ready as u8,
201            Ordering::AcqRel,
202            Ordering::Acquire,
203        )
204        .is_ok()
205}

207/// Claims a ready request for the single matching accepted client socket.
208#[cfg(any(target_os = "windows", test))]
209fn claim_ready_attempt(attempt: &ConditionalAttempt) -> bool {
210    attempt
211        .state
212        .compare_exchange(
213            ConditionalAttemptState::Ready as u8,
214            ConditionalAttemptState::Claimed as u8,
215            Ordering::AcqRel,
216            Ordering::Acquire,
217        )
218        .is_ok()
219}

376    pub cleanup_interval: Duration,

378    pub idle_ttl: Duration,

385impl ProxyConfig {
386    /// Rejects wildcard listeners, missing credentials/endpoints, and zero UDP
387    /// idle time before any sockets are opened.
388    pub fn validate(&self) -> Result<(), ProxyError> {
389        if self.listen.ip().is_unspecified() {
390            return Err(ProxyError::InvalidConfiguration(
391                "listen address must be specific so UDP tuples identify the local destination"
392                    .into(),
393            ));
394        }
395        if self.control_endpoint.is_empty() {
396            return Err(ProxyError::InvalidConfiguration(
397                "control endpoint is required".into(),
398            ));
399        }
400        #[cfg(not(feature = "tls-rustls"))]
401        if self.psk_identity.is_empty() || self.psk_secret.is_empty() {
402            return Err(ProxyError::InvalidConfiguration(
403                "PSK identity and secret are required".into(),
404            ));
405        }
406        if self.udp_idle_timeout.is_zero() {
407            return Err(ProxyError::InvalidConfiguration(
408                "UDP idle timeout must be nonzero".into(),
409            ));
410        }
411        if self.listen_backlog == 0 || self.listen_backlog > i32::MAX as u32 {
412            return Err(ProxyError::InvalidConfiguration(
413                "native listen backlog must be between 1 and i32::MAX; it does not size internal conditional-accept queues".into(),
414            ));
415        }
416        for (name, duration) in [
417            ("UDP idle timeout", self.udp_idle_timeout),
418            ("cleanup interval", self.cleanup_interval),
419            ("idle TTL", self.idle_ttl),
420            ("TCP terminal grace", self.tcp_terminal_grace),
421        ] {
422            duration_to_nanos(duration, name)?;
423        }
424        if self.cleanup_interval.is_zero()
425            || self.idle_ttl.is_zero()
426            || self.tcp_terminal_grace.is_zero()
427            || self.flow_scan_batch == 0
428            || self.flow_scan_batch > 10_000
429        {
430            return Err(ProxyError::InvalidConfiguration(
431                "maintenance settings must be nonzero".into(),
432            ));
433        }
434        Ok(())
435    }
436
437    /// Validates the legacy OpenSSL TLS-PSK credentials used by the PSK
438    /// control client.
439    #[cfg(feature = "tls-psk")]
440    pub fn validate_tls_psk(&self) -> Result<(), ProxyError> {
441        if self.psk_identity.is_empty() || self.psk_secret.is_empty() {
442            return Err(ProxyError::InvalidConfiguration(
443                "PSK identity and secret are required".into(),
444            ));
445        }
446        Ok(())
447    }

457/// Runs a control operation with a local deadline and cooperative shutdown.
458async fn control_operation<T>(
459    shutdown: &mut watch::Receiver<bool>,
460    operation: impl Future<Output = Result<T, ProxyError>>,
461) -> Result<Option<T>, ProxyError> {
462    tokio::select! {
463        _ = shutdown.changed() => Ok(None),
464        result = time::timeout(CONTROL_RPC_TIMEOUT, operation) => {
465            match result {
466                Ok(result) => result.map(Some),
467                Err(_) => Err(ProxyError::Control("control RPC timed out".into())),
468            }
469        }
470    }
471}

473/// Runs serialized host-owned flow maintenance until shutdown.
474#[allow(clippy::too_many_arguments)]
475async fn run_maintenance<C: MappingClient + FlowClient + 'static>(
476    client: Arc<C>,
477    associations: Arc<UdpAssociations<C>>,
478    cleanup_interval: Duration,
479    udp_idle_timeout: Duration,
480    idle_ttl: Duration,
481    tcp_terminal_grace: Duration,
482    flow_scan_batch: u32,
483    mut shutdown: watch::Receiver<bool>,
484) {
485    let initial_retry_delay = cleanup_interval.min(Duration::from_millis(100));
486    let mut retry_delay = initial_retry_delay;
487    let mut pending_deletes = std::collections::HashMap::<(u64, u32), FlowRecord>::new();
488    let udp_idle_timeout_ns =
489        duration_to_nanos(udp_idle_timeout, "UDP idle timeout").expect("validated configuration");
490    let idle_ttl_ns = duration_to_nanos(idle_ttl, "idle TTL").expect("validated configuration");
491    let tcp_terminal_grace_ns = duration_to_nanos(tcp_terminal_grace, "TCP terminal grace")
492        .expect("validated configuration");
493    let mut next_run = time::Instant::now();
494    'maintenance: loop {
495        let sleep = time::sleep_until(next_run);
496        tokio::pin!(sleep);
497        tokio::select! {
498            _ = shutdown.changed() => break,
499            _ = &mut sleep => {
500                let mut token = Vec::new();
501                let mut pass_failed = false;
502                loop {
503                    let page = control_operation(
504                        &mut shutdown,
505                        client.enumerate_flows(token, flow_scan_batch),
506                    )
507                    .await;
508                    let page = match page {
509                        Ok(Some(page)) => page,
510                        Ok(None) => break 'maintenance,
511                        Err(error) => {
512                            tracing::warn!(%error, "host maintenance flow enumeration failed");
513                            pass_failed = true;
514                            break;
515                        }
516                    };
517                    let (flows, next) = page;
518                    for flow in flows {
519                        let age = flow.observed_now_ns.saturating_sub(flow.last_used_ns);
520                        let tcp = flow.original.protocol == TCP_PROTOCOL;
521                        if flow.original.protocol == UDP_PROTOCOL {
522                            associations
523                                .observe_flow(&flow.synthetic, flow.flow_id, flow.generation)
524                                .await;
525                        }
526                        let terminal = tcp
527                            && flow.fin_seen_mask == 0b11
528                            && flow.fin_ack_seen_mask == 0b11;
529                        let idle_threshold = if flow.original.protocol == UDP_PROTOCOL {
530                            udp_idle_timeout_ns
531                        } else {
532                            idle_ttl_ns
533                        };
534                        let expired = flow.tcp_state_flags & (1 << 4) != 0
535                            || (!terminal && age >= idle_threshold)
536                            || (terminal && age >= tcp_terminal_grace_ns);
537                        if expired {
538                            pending_deletes.insert((flow.flow_id, flow.generation), flow);
539                        }
540                    }
541                    if next.is_empty() {
542                        break;
543                    }
544                    token = next;
545                }
546                let pending = pending_deletes.values().cloned().collect::<Vec<_>>();
547                for flow in pending {
548                    match control_operation(
549                        &mut shutdown,
550                        client.delete_flow(
551                            flow.flow_id,
552                            flow.generation,
553                            flow.last_used_ns,
554                        ),
555                    )
556                    .await
557                    {
558                        Ok(Some(report)) => {
559                            if matches!(
560                                report.outcome,
561                                FlowDeleteOutcome::Complete
562                                    | FlowDeleteOutcome::AlreadyAbsent
563                                    | FlowDeleteOutcome::StaleGeneration
564                                    | FlowDeleteOutcome::ObservationMismatch
565                            ) {
566                                pending_deletes.remove(&(flow.flow_id, flow.generation));
567                                if flow.original.protocol == UDP_PROTOCOL
568                                    && matches!(
569                                        report.outcome,
570                                        FlowDeleteOutcome::Complete
571                                            | FlowDeleteOutcome::AlreadyAbsent
572                                    )
573                                {
574                                    associations
575                                        .invalidate(
576                                            &flow.synthetic,
577                                            flow.flow_id,
578                                            flow.generation,
579                                        )
580                                        .await;
581                                }
582                            }
583                            if report.retryable {
584                                pass_failed = true;
585                            }
586                            tracing::info!(
587                                flow_id = flow.flow_id,
588                                generation = flow.generation,
589                                outcome = ?report.outcome,
590                                "host maintenance flow deletion"
591                            );
592                        }
593                        Ok(None) => break 'maintenance,
594                        Err(error) => {
595                            pass_failed = true;
596                            tracing::warn!(
597                                flow_id = flow.flow_id,
598                                generation = flow.generation,
599                                %error,
600                                "host maintenance flow deletion failed"
601                            );
602                        }
603                    }
604                }
605                if pass_failed {
606                    let delay = retry_delay;
607                    retry_delay = retry_delay
608                        .saturating_mul(2)
609                        .min(cleanup_interval);
610                    next_run = time::Instant::now() + delay;
611                } else {
612                    retry_delay = initial_retry_delay;
613                    next_run = time::Instant::now() + cleanup_interval;
614                }
615            }
616        }
617    }
618}

660    /// Binds TCP and UDP listeners at the configured specific address.
661    pub async fn bind(&self) -> Result<(BoundTcpListener, Arc<UdpSocket>), ProxyError> {
662        #[cfg(not(target_os = "windows"))]
663        {
664            let tcp_listener = TcpListener::bind(self.config.listen).await?;
665            let actual_listen = tcp_listener.local_addr()?;
666            let udp_socket = Arc::new(UdpSocket::bind(actual_listen).await?);
667            Ok((
668                BoundTcpListener {
669                    listener: tcp_listener,
670                },
671                udp_socket,
672            ))
673        }
674        #[cfg(target_os = "windows")]
675        {
676            let tcp_listener = windows_conditional::ConditionalListener::bind(
677                self.config.listen,
678                self.config.listen_backlog,
679            )?;
680            let actual_listen = tcp_listener.local_addr();
681            let udp_socket = Arc::new(UdpSocket::bind(actual_listen).await?);
682            Ok((
683                BoundTcpListener {
684                    listener: tcp_listener,
685                },
686                udp_socket,
687            ))
688        }

691    /// Binds both transports, runs them until shutdown or task failure, then
692    /// aborts the sibling task and clears UDP associations.
693    pub async fn run(self, shutdown: watch::Receiver<bool>) -> Result<(), ProxyError> {
694        let (tcp_listener, udp_socket) = self.bind().await?;
695        self.run_bound(tcp_listener, udp_socket, shutdown).await
696    }

698    /// Runs with pre-bound listeners so control-plane activation can safely
699    /// target a listening proxy before any redirected traffic is enabled.
700    pub async fn run_bound(
701        self,
702        tcp_listener: BoundTcpListener,
703        udp_socket: Arc<UdpSocket>,
704        mut shutdown: watch::Receiver<bool>,
705    ) -> Result<(), ProxyError> {
706        // A local fanout makes a sibling task failure a coordinated shutdown
707        // rather than an abort that could strand Windows native threads.
708        let (stop, task_shutdown) = watch::channel(false);
709        let udp = Arc::new(UdpAssociations::new(
710            udp_socket,
711            self.client.clone(),
712            self.config.udp_idle_timeout,
713            task_shutdown.clone(),
714        ));
715        let maintenance = tokio::spawn(run_maintenance(
716            self.client.clone(),
717            udp.clone(),
718            self.config.cleanup_interval,
719            self.config.udp_idle_timeout,
720            self.config.idle_ttl,
721            self.config.tcp_terminal_grace,
722            self.config.flow_scan_batch,
723            task_shutdown.clone(),
724        ));
725        let mut tcp_task = tokio::spawn(run_tcp(
726            tcp_listener,
727            self.client.clone(),
728            task_shutdown.clone(),
729        ));
730        let mut udp_task = tokio::spawn(run_udp(udp.clone(), task_shutdown));
731        let result = tokio::select! {
732            _ = shutdown.changed() => {
733                let _ = stop.send(true);
734                let tcp_result = tcp_task
735                    .await
736                    .map_err(|error| ProxyError::Control(error.to_string()))
737                    .and_then(|result| result);
738                let _ = udp_task.await;
739                let _ = maintenance.await;
740                tcp_result
741            },
742            result = &mut tcp_task => {
743                let _ = stop.send(true);
744                let _ = udp_task.await;
745                let _ = maintenance.await;
746                result
747                    .map_err(|error| ProxyError::Control(error.to_string()))
748                    .and_then(|result| result)
749            },
750            result = &mut udp_task => {
751                let _ = stop.send(true);
752                let _ = tcp_task.await;
753                let _ = maintenance.await;
754                result.map_err(|error| ProxyError::Control(error.to_string()))
755            },
756        };
757        udp.shutdown().await;
758        tracing::info!(
759            protocol = "proxy",
760            reason = "forwarding_stopped",
761            "host proxy shutdown"
762        );
763        result
764    }
765}

767/// Accepts TCP sessions until shutdown and joins or aborts all bridges.
768async fn run_tcp<C: MappingClient + 'static>(
769    listener: BoundTcpListener,
770    client: Arc<C>,
771    shutdown: watch::Receiver<bool>,
772) -> Result<(), ProxyError> {
773    #[cfg(not(target_os = "windows"))]
774    let BoundTcpListener { listener } = listener;
775    #[cfg(not(target_os = "windows"))]
776    {
777        run_tcp_tokio(listener, client, shutdown).await;
778        Ok(())
779    }
780    #[cfg(target_os = "windows")]
781    {
782        windows_conditional::run_tcp_conditional(listener.listener, client, shutdown).await
783    }
784}

786/// Accepts TCP sessions through Tokio on platforms that do not use Windows
787/// conditional accept.
788#[cfg(not(target_os = "windows"))]
789async fn run_tcp_tokio<C: MappingClient + 'static>(
790    listener: TcpListener,
791    client: Arc<C>,
792    mut shutdown: watch::Receiver<bool>,
793) {
794    let mut sessions = JoinSet::new();
795    loop {
796        tokio::select! {
797            _ = shutdown.changed() => break,
798            Some(_) = sessions.join_next(), if !sessions.is_empty() => {}
799            accepted = listener.accept() => {
800                let (stream, _) = match accepted {
801                    Ok(value) => value,
802                    Err(error) => {
803                        tracing::error!(%error, "TCP accept failed");
804                        continue;
805                    }
806                };
807                let client = client.clone();
808                sessions.spawn(async move {
809                    if let Err(error) = bridge_tcp(stream, client).await {
810                        tracing::warn!(%error, "TCP forwarding session failed");
811                    }
812                });
813            }
814        }
815    }
816    sessions.abort_all();
817    while sessions.join_next().await.is_some() {}
818}

820/// Emits a termination event if a running TCP bridge is cancelled.
821struct TcpForwardingGuard {
822    /// Synthetic tuple associated with the bridge.
823    tuple: Tuple,
824    /// Original destination used by the bridge.
825    original_destination: SocketAddr,
826    /// Indicates that the bridge emitted its normal termination event.
827    completed: bool,
828}
829
830impl Drop for TcpForwardingGuard {
831    /// Reports cancellation when the bridge future is dropped before completion.
832    fn drop(&mut self) {
833        if !self.completed {
834            tracing::info!(
835                protocol = "tcp",
836                synthetic_source = %self.tuple.source,
837                synthetic_destination = %self.tuple.destination,
838                original_destination = %self.original_destination,
839                reason = "cancelled",
840                "TCP forwarding terminated"
841            );
842        }
843    }
844}

846/// Looks up the accepted tuple, connects to the original destination, and
847/// copies bytes bidirectionally until either stream closes.
848#[cfg(any(not(target_os = "windows"), test))]
849async fn bridge_tcp<C: MappingClient + 'static>(
850    accepted: TcpStream,
851    client: Arc<C>,
852) -> Result<(), ProxyError> {
853    let tuple = Tuple {
854        source: accepted.peer_addr()?,
855        destination: accepted.local_addr()?,
856        protocol: TCP_PROTOCOL,
857    };
858    let original = match client.get_mapping(&tuple).await {
859        Ok(original) => original,
860        Err(error) => {
861            tracing::warn!(
862                protocol = "tcp",
863                synthetic_source = %tuple.source,
864                synthetic_destination = %tuple.destination,
865                error = %error,
866                "TCP mapping lookup failed"
867            );
868            return Err(error);
869        }
870    };
871    validate_tcp_mapping(&tuple, &original)?;
872    let outbound = match TcpStream::connect(original.address).await {
873        Ok(outbound) => outbound,
874        Err(error) => {
875            tracing::warn!(
876                protocol = "tcp",
877                synthetic_source = %tuple.source,
878                synthetic_destination = %tuple.destination,
879                original_destination = %original.address,
880                error = %error,
881                "TCP outbound connection failed"
882            );
883            return Err(error.into());
884        }
885    };
886    bridge_tcp_streams(accepted, outbound, tuple, original.address).await
887}

889/// Validates a TCP mapping before either forwarding path obtains an outbound
890/// socket. This protects the generic mapping-client interface as well as the
891/// protocol client's protobuf validation.
892fn validate_tcp_mapping(tuple: &Tuple, original: &OriginalDestination) -> Result<(), ProxyError> {
893    if original.protocol != TCP_PROTOCOL {
894        return Err(ProxyError::InvalidMapping(
895            "TCP lookup returned a non-TCP mapping".into(),
896        ));
897    }
898    if original.address.ip().is_unspecified()
899        || original.address.port() == 0
900        || original.address.is_ipv4() != tuple.destination.is_ipv4()
901    {
902        return Err(ProxyError::InvalidMapping(
903            "TCP mapping destination has an invalid address, port, or family".into(),
904        ));
905    }
906    Ok(())
907}

909/// Bridges a client socket and an already validated, connected outbound socket.
910/// Windows conditional accept uses this entry point so the socket that caused
911/// `CF_ACCEPT` is never connected a second time.
912async fn bridge_tcp_streams(
913    mut accepted: TcpStream,
914    mut outbound: TcpStream,
915    tuple: Tuple,
916    original_destination: SocketAddr,
917) -> Result<(), ProxyError> {
918    let mut termination = TcpForwardingGuard {
919        tuple: tuple.clone(),
920        original_destination,
921        completed: false,
922    };
923    tracing::info!(
924        protocol = "tcp",
925        synthetic_source = %tuple.source,
926        synthetic_destination = %tuple.destination,
927        original_destination = %original_destination,
928        reason = "forwarding_started",
929        "TCP forwarding started"
930    );
931    match io::copy_bidirectional(&mut accepted, &mut outbound).await {
932        Ok((client_to_destination, destination_to_client)) => {
933            tracing::info!(
934                protocol = "tcp",
935                synthetic_source = %tuple.source,
936                synthetic_destination = %tuple.destination,
937                original_destination = %original_destination,
938                client_to_destination,
939                destination_to_client,
940                reason = "stream_closed",
941                "TCP forwarding terminated"
942            );
943            termination.completed = true;
944        }
945        Err(error) => {
946            tracing::warn!(
947                protocol = "tcp",
948                synthetic_source = %tuple.source,
949                synthetic_destination = %tuple.destination,
950                original_destination = %original_destination,
951                error = %error,
952                reason = "stream_error",
953                "TCP forwarding terminated with error"
954            );
955            termination.completed = true;
956            return Err(error.into());
957        }
958    }
959    Ok(())
960}

962/// Receives datagrams, resolves/creates associations, and relays replies.
963async fn run_udp<C: MappingClient + 'static>(
964    associations: Arc<UdpAssociations<C>>,
965    mut shutdown: watch::Receiver<bool>,
966) {
967    let mut buffer = vec![0_u8; 65_535];
968    loop {
969        tokio::select! {
970            _ = shutdown.changed() => break,
971            received = associations.socket.recv_from(&mut buffer) => {
972                let (length, client_address) = match received {
973                    Ok(value) => value,
974                    Err(error) => {
975                        tracing::error!(%error, "UDP receive failed");
976                        continue;
977                    }
978                };
979                tokio::select! {
980                    _ = shutdown.changed() => break,
981                    result = associations.forward(client_address, &buffer[..length]) => {
982                        if let Err(error) = result {
983                            associations.report_failure(&error).await;
984                        }
985                    }
986                }
987            }
988        }
989    }
990}

992/// Shared UDP association table and lifecycle coordination state.
993type AssociationTable = HashMap<Tuple, Arc<UdpAssociation>>;

995/// Coordinates cached UDP associations and their relay lifetimes.
996struct UdpAssociations<C> {
997    /// Client-facing UDP socket bound to the proxy listener.
998    socket: Arc<UdpSocket>,
999    /// Mapping service used to resolve each client tuple.
1000    client: Arc<C>,
1001    /// Associations keyed by client tuple.
1002    entries: Arc<Mutex<AssociationTable>>,
1003    /// Lifetime used by relay timeouts and periodic reaping.
1004    idle_timeout: Duration,
1005    /// Watch receiver used to stop relay tasks.
1006    shutdown: watch::Receiver<bool>,
1007    /// Timestamp used to rate-limit repeated forwarding warnings.
1008    last_failure_log: Mutex<Option<std::time::Instant>>,
1009}

1011/// One connected outbound UDP socket paired with its originating client.
1012struct UdpAssociation {
1013    /// Synthetic tuple identifying this cache entry.
1014    key: Tuple,
1015    /// Weak cache reference used when the relay expires on its own.
1016    entries: Weak<Mutex<AssociationTable>>,
1017    /// Client endpoint receiving relayed responses.
1018    client_address: SocketAddr,
1019    /// Synthetic client tuple used for the control-service mapping.
1020    synthetic_source: SocketAddr,
1021    /// Synthetic proxy tuple used for the control-service mapping.
1022    synthetic_destination: SocketAddr,
1023    /// Current mapped destination; changes cause association replacement.
1024    destination: SocketAddr,
1025    /// Monotonic creation time used to report association lifetime.
1026    created_at: std::time::Instant,
1027    /// Connected UDP socket used for outbound datagrams and replies.
1028    outbound: Arc<UdpSocket>,
1029    /// Last successful send or receive time for idle reaping.
1030    last_seen: Mutex<std::time::Instant>,
1031    /// Wakes the relay when client-to-destination activity extends its lease.
1032    activity: Notify,
1033    /// Dataplane flow incarnation associated with this relay, when observed.
1034    flow_identity: Mutex<Option<(u64, u32)>>,
1035    /// Signals this relay to stop when the dataplane flow is deleted.
1036    cancel: watch::Sender<bool>,
1037    /// Becomes true when the relay has released its socket.
1038    relay_done: watch::Sender<bool>,
1039}

1041/// Extends an association lease after successful traffic in either direction.
1042async fn touch_association(association: &UdpAssociation) {
1043    *association.last_seen.lock().await = std::time::Instant::now();
1044    association.activity.notify_one();
1045}

1065    /// Reuses an active association, resolving only when it is absent or a
1066    /// destination-specific send failure requires one retry.
1067    async fn forward(&self, client_address: SocketAddr, payload: &[u8]) -> Result<(), ProxyError> {
1068        let tuple = Tuple {
1069            source: client_address,
1070            destination: self.socket.local_addr()?,
1071            protocol: UDP_PROTOCOL,
1072        };
1073        let association = self.entries.lock().await.get(&tuple).cloned();
1074        let association = match association {
1075            Some(association) => association,
1076            None => self.resolve_association(&tuple, client_address).await?,
1077        };
1078        if let Err(error) = association.outbound.send(payload).await {
1079            tracing::warn!(
1080                protocol = "udp",
1081                synthetic_source = %tuple.source,
1082                synthetic_destination = %tuple.destination,
1083                original_destination = %association.destination,
1084                error = %error,
1085                "UDP datagram send failed"
1086            );
1087            if !matches!(
1088                error.kind(),
1089                io::ErrorKind::ConnectionRefused
1090                    | io::ErrorKind::TimedOut
1091                    | io::ErrorKind::NetworkUnreachable
1092                    | io::ErrorKind::HostUnreachable
1093            ) {
1094                return Err(error.into());
1095            }
1096            let old = self.entries.lock().await.remove(&tuple);
1097            if let Some(old) = old {
1098                Self::stop_association(old).await;
1099            }
1100            let replacement = self.resolve_association(&tuple, client_address).await?;
1101            replacement.outbound.send(payload).await?;
1102            touch_association(&replacement).await;
1103            return Ok(());
1104        }
1105        touch_association(&association).await;
1106        Ok(())
1107    }

1109    /// Performs the one mapping lookup needed to create an association.
1110    async fn resolve_association(
1111        &self,
1112        tuple: &Tuple,
1113        client_address: SocketAddr,
1114    ) -> Result<Arc<UdpAssociation>, ProxyError> {
1115        let mapping = self.client.get_mapping(tuple).await.map_err(|error| {
1116            tracing::warn!(
1117                protocol = "udp",
1118                synthetic_source = %tuple.source,
1119                synthetic_destination = %tuple.destination,
1120                error = %error,
1121                "UDP mapping lookup failed"
1122            );
1123            error
1124        })?;
1125        if mapping.protocol != UDP_PROTOCOL {
1126            tracing::warn!(
1127                protocol = "udp",
1128                synthetic_source = %tuple.source,
1129                synthetic_destination = %tuple.destination,
1130                original_destination = %mapping.address,
1131                reason = "protocol_mismatch",
1132                "UDP mapping validation failed"
1133            );
1134            return Err(ProxyError::InvalidMapping(
1135                "UDP lookup returned a non-UDP mapping".into(),
1136            ));
1137        }
1138        self.insert_association(tuple.clone(), client_address, mapping, None)
1139            .await
1140    }

1142    /// Binds and connects a new outbound UDP socket, publishing it atomically
1143    /// if another task has not already installed the same destination.
1144    async fn insert_association(
1145        &self,
1146        tuple: Tuple,
1147        client_address: SocketAddr,
1148        mapping: OriginalDestination,
1149        previous_destination: Option<SocketAddr>,
1150    ) -> Result<Arc<UdpAssociation>, ProxyError> {
1151        let destination = mapping.address;
1152        let outbound = Arc::new(match UdpSocket::bind(unspecified_for(destination)).await {
1153            Ok(outbound) => outbound,
1154            Err(error) => {
1155                tracing::warn!(
1156                    protocol = "udp",
1157                    synthetic_source = %tuple.source,
1158                    synthetic_destination = %tuple.destination,
1159                    original_destination = %destination,
1160                    error = %error,
1161                    "UDP outbound socket bind failed"
1162                );
1163                return Err(error.into());
1164            }
1165        });
1166        if let Err(error) = outbound.connect(destination).await {
1167            tracing::warn!(
1168                protocol = "udp",
1169                synthetic_source = %tuple.source,
1170                synthetic_destination = %tuple.destination,
1171                original_destination = %destination,
1172                error = %error,
1173                "UDP outbound connection failed"
1174            );
1175            return Err(error.into());
1176        }
1177        let (cancel, relay_shutdown) = watch::channel(false);
1178        let (relay_done, _) = watch::channel(false);
1179        let candidate = Arc::new(UdpAssociation {
1180            key: tuple.clone(),
1181            entries: Arc::downgrade(&self.entries),
1182            client_address,
1183            synthetic_source: tuple.source,
1184            synthetic_destination: tuple.destination,
1185            destination,
1186            created_at: std::time::Instant::now(),
1187            outbound,
1188            last_seen: Mutex::new(std::time::Instant::now()),
1189            activity: Notify::new(),
1190            flow_identity: Mutex::new(None),
1191            cancel,
1192            relay_done,
1193        });
1194        let entries = self.entries.lock().await;
1195        if let Some(existing) = entries.get(&tuple) {
1196            if existing.destination == destination {
1197                return Ok(existing.clone());
1198            }
1199        }
1200        drop(entries);
1201        spawn_udp_relay(
1202            candidate.clone(),
1203            self.socket.clone(),
1204            self.idle_timeout,
1205            self.shutdown.clone(),
1206            relay_shutdown,
1207            candidate.relay_done.clone(),
1208        );
1209        let mut entries = self.entries.lock().await;
1210        if let Some(existing) = entries.get(&tuple) {
1211            if existing.destination == destination {
1212                let existing = existing.clone();
1213                drop(entries);
1214                Self::stop_association(candidate).await;
1215                return Ok(existing);
1216            }
1217        }
1218        let source = tuple.source;
1219        let proxy = tuple.destination;
1220        let previous = entries.insert(tuple, candidate.clone());
1221        drop(entries);
1222        if let Some(previous) = previous {
1223            Self::stop_association(previous).await;
1224        }
1225        match previous_destination {
1226            Some(previous_destination) => tracing::info!(
1227                protocol = "udp",
1228                synthetic_source = %source,
1229                synthetic_destination = %proxy,
1230                previous_destination = %previous_destination,
1231                original_destination = %destination,
1232                association_age_ms = 0_u64,
1233                reason = "association_replaced",
1234                "UDP association replaced"
1235            ),
1236            None => tracing::info!(
1237                protocol = "udp",
1238                synthetic_source = %source,
1239                synthetic_destination = %proxy,
1240                original_destination = %destination,
1241                association_age_ms = 0_u64,
1242                reason = "association_created",
1243                "UDP association created"
1244            ),
1245        }
1246        Ok(candidate)
1247    }

1249    /// Logs forwarding failures at most once per second.
1250    async fn report_failure(&self, error: &ProxyError) {

1283    /// Removes a locally cached association after confirmed host deletion.
1284    async fn invalidate(&self, tuple: &Tuple, flow_id: u64, generation: u32) {
1285        let association = self.entries.lock().await.get(tuple).cloned();
1286        let Some(association) = association else {
1287            return;
1288        };
1289        if *association.flow_identity.lock().await != Some((flow_id, generation)) {
1290            return;
1291        }
1292        let association = {
1293            let mut entries = self.entries.lock().await;
1294            entries
1295                .get(tuple)
1296                .is_some_and(|current| Arc::ptr_eq(current, &association))
1297                .then(|| entries.remove(tuple))
1298                .flatten()
1299        };
1300        if let Some(association) = association {
1301            Self::stop_association(association).await;
1302            tracing::info!(
1303                protocol = "udp",
1304                synthetic_source = %tuple.source,
1305                synthetic_destination = %tuple.destination,
1306                reason = "flow_deleted",
1307                "UDP association invalidated"
1308            );
1309        }
1310    }

1319    /// Cancels an association relay and waits until it has released its socket.
1320    async fn stop_association(association: Arc<UdpAssociation>) {
1321        let mut relay_done = association.relay_done.subscribe();
1322        if *relay_done.borrow() {
1323            return;
1324        }
1325        let _ = association.cancel.send(true);
1326        while !*relay_done.borrow() {
1327            if relay_done.changed().await.is_err() {
1328                return;
1329            }
1330        }
1331    }

1335async fn finish_udp_relay(association: &Arc<UdpAssociation>, relay_done: &watch::Sender<bool>) {
1336    if let Some(entries) = association.entries.upgrade() {

1339            .get(&association.key)
1340            .is_some_and(|current| Arc::ptr_eq(current, association))
1341        {
1342            entries.remove(&association.key);
1343        }
1344    }
1345    let _ = relay_done.send(true);

1350fn spawn_udp_relay(
1351    association: Arc<UdpAssociation>,
1352    client_socket: Arc<UdpSocket>,
1353    idle_timeout: Duration,
1354    mut shutdown: watch::Receiver<bool>,
1355    mut cancel: watch::Receiver<bool>,
1356    relay_done: watch::Sender<bool>,
1357) {
1358    tokio::spawn(async move {
1359        let mut buffer = vec![0_u8; 65_535];
1360        loop {
1361            let deadline = *association.last_seen.lock().await + idle_timeout;
1362            let idle = time::sleep_until(deadline.into());
1363            let activity = association.activity.notified();
1364            tokio::pin!(idle);
1365            tokio::pin!(activity);
1366            let result = tokio::select! {
1367                _ = shutdown.changed() => {
1368                    tracing::info!(
1369                        protocol = "udp",
1370                        synthetic_source = %association.synthetic_source,
1371                        synthetic_destination = %association.synthetic_destination,
1372                        client_address = %association.client_address,
1373                        original_destination = %association.destination,
1374                        association_age_ms = association.created_at.elapsed().as_millis() as u64,
1375                        reason = "proxy_shutdown",
1376                        "UDP relay stopped"
1377                    );
1378                    finish_udp_relay(&association, &relay_done).await;
1379                    return;
1380                },
1381                _ = cancel.changed() => {
1382                    finish_udp_relay(&association, &relay_done).await;
1383                    return;
1384                },
1385                _ = &mut activity => continue,
1386                _ = &mut idle => {
1387                    if std::time::Instant::now()
1388                        .saturating_duration_since(*association.last_seen.lock().await)
1389                        < idle_timeout
1390                    {
1391                        continue;
1392                    }
1393                    Err(())
1394                },
1395                result = association.outbound.recv(&mut buffer) => Ok(result),
1396            };
1397            match result {
1398                Ok(Ok(length)) => {
1399                    if let Err(error) = client_socket
1400                        .send_to(&buffer[..length], association.client_address)
1401                        .await
1402                    {

1404                            protocol = "udp",
1405                            synthetic_source = %association.synthetic_source,
1406                            synthetic_destination = %association.synthetic_destination,

1410                            association_age_ms = association.created_at.elapsed().as_millis() as u64,
1411                            reason = "relay_delivery",
1412                            "UDP relay delivery failed"
1413                        );
1414                        tracing::info!(
1415                            protocol = "udp",
1416                            synthetic_source = %association.synthetic_source,
1417                            synthetic_destination = %association.synthetic_destination,
1418                            client_address = %association.client_address,
1419                            original_destination = %association.destination,
1420                            association_age_ms = association.created_at.elapsed().as_millis() as u64,
1421                            reason = "delivery_error",
1422                            "UDP relay stopped"
1423                        );
1424                        break;
1425                    }
1426                    touch_association(&association).await;
1427                }
1428                Ok(Err(error)) => {
1429                    tracing::warn!(
1430                        protocol = "udp",
1431                        synthetic_source = %association.synthetic_source,
1432                        synthetic_destination = %association.synthetic_destination,
1433                        client_address = %association.client_address,
1434                        original_destination = %association.destination,
1435                        error = %error,
1436                        association_age_ms = association.created_at.elapsed().as_millis() as u64,
1437                        reason = "receive_error",
1438                        "UDP relay receive failed"
1439                    );
1440                    tracing::info!(
1441                        protocol = "udp",
1442                        synthetic_source = %association.synthetic_source,
1443                        synthetic_destination = %association.synthetic_destination,
1444                        client_address = %association.client_address,
1445                        original_destination = %association.destination,
1446                        association_age_ms = association.created_at.elapsed().as_millis() as u64,
1447                        reason = "receive_error",
1448                        "UDP relay stopped"
1449                    );
1450                    break;
1451                }
1452                Err(()) => {
1453                    tracing::info!(
1454                        protocol = "udp",
1455                        synthetic_source = %association.synthetic_source,
1456                        synthetic_destination = %association.synthetic_destination,
1457                        client_address = %association.client_address,
1458                        original_destination = %association.destination,
1459                        idle_timeout_ms = idle_timeout.as_millis() as u64,
1460                        association_age_ms = association.created_at.elapsed().as_millis() as u64,
1461                        reason = "idle_timeout",
1462                        "UDP relay stopped"
1463                    );
1464                    break;
1465                }
1466            }
1467        }
1468        finish_udp_relay(&association, &relay_done).await;

1742        /// exposing callback-owned storage across an await point.
1743        fn deferred_attempt(&self) -> Option<(ConditionalRequestId, std::time::Instant)> {
1744            let attempt = self.attempt.lock().expect("conditional attempt poisoned");
1745            attempt.as_ref().and_then(|slot| {

1747                    .then(|| (slot.attempt.id.clone(), slot.attempt.deferred_at))

1754            let attempt = self.attempt.lock().expect("conditional attempt poisoned");
1755            attempt
1756                .as_ref()

1762        /// attempt under the same lock used by the callback makes this store
1763        /// the shutdown linearization point: a later callback cannot claim a
1764        /// ready socket.
1765        fn stop_and_reject_pending(&self) {
1766            let slot = {
1767                let mut attempt = self.attempt.lock().expect("conditional attempt poisoned");
1768                self.accepting.store(false, Ordering::Release);
1769                attempt.take()
1770            };
1771            if let Some(slot) = slot {
1772                slot.attempt
1773                    .state
1774                    .store(ConditionalAttemptState::Cancelled as u8, Ordering::Release);
1775                self.admissions.release(&slot.attempt);
1776            }
1777            self.release_handoff_reservation();
1778            self.wake();
1779        }

1782        fn reserve_handoff(&self) -> bool {
1783            let mut current = self.handoffs.load(Ordering::Acquire);
1784            loop {
1785                if current >= self.admissions.limit {
1786                    return false;
1787                }
1788                match self.handoffs.compare_exchange_weak(
1789                    current,
1790                    current + 1,
1791                    Ordering::AcqRel,
1792                    Ordering::Acquire,
1793                ) {
1794                    Ok(_) => {
1795                        self.handoff_reserved.store(true, Ordering::Release);
1796                        return true;
1797                    }
1798                    Err(observed) => current = observed,
1799                }
1800            }
1801        }

1807        }

1810        /// into a handoff owner.

1814            }
1815        }

1820            slot.attempt.state.store(state as u8, Ordering::Release);

1836                    .then(|| attempt.take())
1837                    .flatten()
1838            };

1842        }

1900        /// Binds and configures a TCP socket for conditional accept before
1901        /// native `listen`, then transfers it to the coordinator.
1902        pub(super) fn bind(local_addr: SocketAddr, backlog: u32) -> Result<Self, ProxyError> {
1903            let socket = match local_addr {
1904                SocketAddr::V4(_) => TcpSocket::new_v4()?,
1905                SocketAddr::V6(_) => TcpSocket::new_v6()?,
1906            };
1907            socket.bind(local_addr)?;
1908            // SAFETY: the event is an unnamed, manual-reset event owned by the
1909            // context and used only to wake its coordinator thread.
1910            let wake_event = unsafe { CreateEventW(ptr::null(), 1, 0, ptr::null()) };
1911            if wake_event.is_null() {
1912                return Err(ProxyError::Io(io::Error::last_os_error()));
1913            }
1914            // Native backlog is deliberately independent from the fixed
1915            // callback/worker capacity, so a valid i32::MAX backlog cannot
1916            // allocate an application-sized channel at startup.
1917            let (request_sender, request_receiver) =
1918                mpsc::sync_channel(CONDITIONAL_ATTEMPT_CAPACITY);
1919            let context = Arc::new(ConditionalContext::new(request_sender, wake_event));
1920            let enabled: c_int = 1;
1921            // SAFETY: `TcpSocket` owns a bound but not-yet-listening socket.
1922            // The option points to `enabled` only for this synchronous call.
1923            if unsafe {
1924                setsockopt(
1925                    socket.as_raw_socket() as Socket,
1926                    SOL_SOCKET,
1927                    SO_CONDITIONAL_ACCEPT,
1928                    (&enabled as *const c_int).cast::<c_char>(),
1929                    mem::size_of_val(&enabled) as c_int,
1930                )
1931            } != 0
1932            {
1933                return Err(last_socket_error("enable SO_CONDITIONAL_ACCEPT"));
1934            }
1935            let listener = socket.listen(backlog)?;
1936            let local_addr = listener.local_addr()?;
1937            let listener = listener.into_std()?;
1938            listener.set_nonblocking(true)?;
1939            let socket = OwnedSocket::new(listener.into_raw_socket() as Socket);
1940            Ok(Self {
1941                socket,
1942                local_addr,
1943                context,
1944                request_receiver,
1945            })
1946        }

1996    /// Starts a worker with a private current-thread runtime. It never calls
1997    /// `Handle::block_on`, so it cannot nest a runtime on Tokio's worker
1998    /// threads.
1999    fn spawn_preconnect_worker<C: MappingClient + 'static>(
2000        client: Arc<C>,
2001        context: Arc<ConditionalContext>,
2002        receiver: Receiver<()>,
2003        sender: mpsc::Sender<WorkerResult>,
2004    ) -> Result<JoinHandle<Result<(), ProxyError>>, ProxyError> {
2005        thread::Builder::new()
2006            .name("ssp-preconnect".into())
2007            .spawn(move || {
2008                let panic_context = context.clone();
2009                match std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
2010                    run_preconnect_worker(client, context, receiver, sender)
2011                })) {
2012                    Ok(result) => result,
2013                    Err(_) => {
2014                        panic_context.stop_and_reject_pending();
2015                        Err(ProxyError::Control(
2016                            "conditional preconnect worker thread panicked".into(),
2017                        ))
2018                    }
2019                }
2020            })
2021            .map_err(ProxyError::Io)
2022    }

2024    /// Runs the preconnect worker and reports runtime construction failures to
2025    /// the async proxy owner instead of treating them as clean shutdown.
2026    fn run_preconnect_worker<C: MappingClient + 'static>(
2027        client: Arc<C>,
2028        context: Arc<ConditionalContext>,
2029        receiver: Receiver<()>,
2030        sender: mpsc::Sender<WorkerResult>,
2031    ) -> Result<(), ProxyError> {
2032        let runtime = Builder::new_current_thread()
2033            .enable_all()
2034            .build()
2035            .map_err(|error| {
2036                context.stop_and_reject_pending();
2037                ProxyError::Control(format!(
2038                    "conditional preconnect runtime initialization failed: {error}"
2039                ))
2040            })?;
2041        while context.accepting.load(Ordering::Acquire) {
2042            match receiver.recv_timeout(Duration::from_millis(10)) {
2043                Ok(()) => {}
2044                Err(mpsc::RecvTimeoutError::Timeout) => continue,
2045                Err(mpsc::RecvTimeoutError::Disconnected) => break,
2046            }
2047            let Some((id, deferred_at)) = context.deferred_attempt() else {
2048                continue;
2049            };
2050            if context.attempt_state(&id) != Some(ConditionalAttemptState::Deferred) {
2051                continue;
2052            }
2053            let result = runtime.block_on(preconnect_attempt(
2054                client.clone(),
2055                context.clone(),
2056                id.clone(),
2057                deferred_at,
2058            ));
2059            let result = match result {
2060                Ok(prepared) => WorkerResult::Ready { id, prepared },
2061                Err(error) => WorkerResult::Rejected { id, error },
2062            };
2063            if sender.send(result).is_err() {
2064                break;
2065            }
2066            context.wake();
2067        }
2068        Ok(())
2069    }

2071    /// Performs the asynchronous mapping lookup and TCP connect entirely
2072    /// outside the condition callback, bounded from the first `CF_DEFER`.
2073    async fn preconnect_attempt<C: MappingClient + 'static>(
2074        client: Arc<C>,
2075        context: Arc<ConditionalContext>,
2076        id: ConditionalRequestId,
2077        deferred_at: std::time::Instant,
2078    ) -> Result<PreparedSocket, ProxyError> {
2079        let remaining = conditional_preconnect_budget(deferred_at)?;
2080        let work = async {
2081            let mapping = client.get_mapping(&id.tuple).await?;
2082            validate_tcp_mapping(&id.tuple, &mapping)?;
2083            let outbound = TcpStream::connect(mapping.address).await?;
2084            let outbound = outbound.into_std()?;
2085            outbound.set_nonblocking(true)?;
2086            Ok(PreparedSocket {
2087                // SAFETY: `into_raw_socket` transfers sole ownership from the
2088                // standard stream to the RAII socket wrapper.
2089                socket: OwnedSocket::new(outbound.into_raw_socket() as Socket),
2090                destination: mapping.address,
2091            })
2092        };
2093        tokio::select! {
2094            result = time::timeout(remaining, work) => {
2095                result
2096                    .map_err(|_| ProxyError::Control("conditional preconnect timed out".into()))?
2097            }
2098            _ = wait_for_attempt_cancellation(context, id.clone()) => Err(ProxyError::Control(
2099                "conditional preconnect cancelled".into(),
2100            )),
2101        }
2102    }

2115    /// Publishes only a current deferred worker completion into its slot.
2116    fn apply_worker_result(context: &ConditionalContext, result: WorkerResult) {
2117        let (id, result) = match result {
2118            WorkerResult::Ready { id, prepared } => (id, Ok(prepared)),
2119            WorkerResult::Rejected { id, error } => (id, Err(error)),
2120        };
2121        let mut attempt = context
2122            .attempt
2123            .lock()
2124            .expect("conditional attempt poisoned");
2125        let Some(slot) = attempt.as_mut().filter(|slot| slot.attempt.id == id) else {
2126            return;
2127        };
2128        if slot.attempt.state() != ConditionalAttemptState::Deferred {
2129            return;
2130        }
2131        match result {
2132            Ok(prepared) => {
2133                slot.prepared = Some(prepared);
2134                if !mark_attempt_ready(&slot.attempt) {
2135                    let _ = slot.prepared.take();
2136                }
2137            }
2138            Err(error) => {
2139                tracing::warn!(
2140                    protocol = "tcp",
2141                    synthetic_source = %id.tuple.source,
2142                    synthetic_destination = %id.tuple.destination,
2143                    error = %error,
2144                    "conditional TCP mapping lookup or preconnect failed"
2145                );
2146                slot.attempt
2147                    .state
2148                    .store(ConditionalAttemptState::Rejected as u8, Ordering::Release);
2149            }
2150        }
2151    }

2153    /// Removes a claimed slot and extracts its paired raw socket owners.
2154    fn take_handoff(
2155        context: &Arc<ConditionalContext>,
2156        generation: u64,
2157        client: OwnedSocket,
2158    ) -> Option<SocketHandoff> {
2159        let slot = {
2160            let mut attempt = context
2161                .attempt
2162                .lock()
2163                .expect("conditional attempts poisoned");
2164            attempt
2165                .as_ref()
2166                .is_some_and(|slot| {
2167                    slot.attempt.id.generation == generation
2168                        && slot.attempt.state() == ConditionalAttemptState::Claimed
2169                })
2170                .then(|| attempt.take())
2171                .flatten()
2172        }?;
2173        let tuple = slot.attempt.id.tuple.clone();
2174        let attempt = slot.attempt;
2175        let outbound = match slot.prepared {
2176            Some(outbound) => outbound,
2177            None => {
2178                context.release_handoff_reservation();
2179                context.admissions.release(&attempt);
2180                return None;
2181            }
2182        };
2183        context.admissions.release(&attempt);
2184        if !context.handoff_reserved.swap(false, Ordering::AcqRel) {
2185            return None;
2186        }
2187        Some(SocketHandoff {
2188            client,
2189            outbound,
2190            tuple,
2191            _reservation: HandoffReservation {
2192                context: context.clone(),
2193            },
2194        })

2197    /// Drives nonblocking `WSAAccept` on its own native thread. The event
2198    /// wakes it promptly for worker completions and shutdown; the bounded
2199    /// polling interval also services incoming connection attempts.
2200    fn run_coordinator(
2201        socket: OwnedSocket,
2202        context: Arc<ConditionalContext>,
2203        result_receiver: Receiver<WorkerResult>,
2204        handoff_sender: tokio_mpsc::Sender<SocketHandoff>,
2205    ) -> Result<(), ProxyError> {
2206        while context.accepting.load(Ordering::Acquire) {
2207            while let Ok(result) = result_receiver.try_recv() {
2208                apply_worker_result(&context, result);
2209            }
2210            context.accepted_generation.store(0, Ordering::Release);
2211            // SAFETY: `socket` remains owned by this thread; the callback
2212            // pointer targets the context Arc retained for the entire loop.
2213            let accepted = unsafe {
2214                WSAAccept(
2215                    socket.raw(),
2216                    ptr::null_mut(),
2217                    ptr::null_mut(),
2218                    Some(condition_callback),
2219                    Arc::as_ptr(&context) as usize,
2220                )
2221            };
2222            if accepted != INVALID_SOCKET {
2223                let client = OwnedSocket::new(accepted);
2224                let generation = context.accepted_generation.load(Ordering::Acquire);
2225                if generation == 0 {
2226                    tracing::error!("WSAAccept returned a socket without a conditional request");
2227                } else if let Some(handoff) = take_handoff(&context, generation, client) {
2228                    if let Err(error) = set_nonblocking(handoff.client.raw()) {
2229                        tracing::warn!(%error, "conditional accepted socket setup failed");
2230                    } else {
2231                        match handoff_sender.try_send(handoff) {
2232                            Ok(()) => {}
2233                            Err(tokio_mpsc::error::TrySendError::Full(handoff)) => {
2234                                drop(handoff);
2235                                tracing::warn!("conditional TCP handoff capacity exhausted");
2236                            }
2237                            Err(tokio_mpsc::error::TrySendError::Closed(handoff)) => {
2238                                drop(handoff);
2239                                break;
2240                            }
2241                        }
2242                    }
2243                } else {
2244                    context.reject_claimed_generation(generation);
2245                    tracing::error!(
2246                        generation,
2247                        "conditional accepted socket had no matching outbound socket"
2248                    );
2249                }
2250            } else {
2251                // A deferred or rejected conditional request normally makes
2252                // WSAAccept fail with one of these transient values.
2253                let error = unsafe { WSAGetLastError() };
2254                let generation = context.accepted_generation.load(Ordering::Acquire);
2255                if generation != 0 {
2256                    context.reject_claimed_generation(generation);
2257                } else if error != WSAEWOULDBLOCK && error != WSATRY_AGAIN {
2258                    context.reject_active_attempt();
2259                }
2260                context.reject_terminal_attempt();
2261                if !is_conditional_client_error(error) {
2262                    context.stop_and_reject_pending();
2263                    return Err(socket_error("WSAAccept", error));
2264                }
2265                if error != WSAEWOULDBLOCK && error != WSATRY_AGAIN {
2266                    tracing::debug!(
2267                        error,
2268                        "WSAAccept rejected or lost a conditional client request"
2269                    );
2270                }
2271            }
2272            // SAFETY: the manual-reset event belongs to the still-live
2273            // context. Resetting after each bounded wait is safe because the
2274            // next iteration also polls both queues.
2275            unsafe {
2276                let result = WaitForSingleObject(context.wake_event, 10);
2277                if result == WAIT_FAILED {
2278                    let error = io::Error::last_os_error();
2279                    context.stop_and_reject_pending();
2280                    return Err(ProxyError::Io(error));
2281                }
2282                if result != WAIT_TIMEOUT && ResetEvent(context.wake_event) == 0 {
2283                    let error = io::Error::last_os_error();
2284                    context.stop_and_reject_pending();
2285                    return Err(ProxyError::Io(error));
2286                }
2287            }
2288        }
2289        context.stop_and_reject_pending();
2290        Ok(())
2291    }

2311    /// Receives native handoffs, converts each pair atomically into Tokio
2312    /// streams, and joins all bridge tasks before returning to `run_bound`.
2313    pub(super) async fn run_tcp_conditional<C: MappingClient + 'static>(
2314        listener: ConditionalListener,
2315        client: Arc<C>,
2316        mut shutdown: watch::Receiver<bool>,
2317    ) -> Result<(), ProxyError> {
2318        let ConditionalListener {
2319            socket,
2320            context,
2321            request_receiver,
2322            ..
2323        } = listener;
2324        let (worker_result_sender, worker_result_receiver) = mpsc::channel();
2325        let worker = spawn_preconnect_worker(
2326            client,
2327            context.clone(),
2328            request_receiver,
2329            worker_result_sender,
2330        )?;
2331        let (handoff_sender, mut handoff_receiver) = tokio_mpsc::channel(context.admissions.limit);
2332        let coordinator_context = context.clone();
2333        let coordinator = match thread::Builder::new()
2334            .name("ssp-conditional-accept".into())
2335            .spawn(move || {
2336                let panic_context = coordinator_context.clone();
2337                match std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
2338                    run_coordinator(
2339                        socket,
2340                        coordinator_context,
2341                        worker_result_receiver,
2342                        handoff_sender,
2343                    )
2344                })) {
2345                    Ok(result) => result,
2346                    Err(_) => {
2347                        panic_context.stop_and_reject_pending();
2348                        Err(ProxyError::Control(
2349                            "conditional coordinator thread panicked".into(),
2350                        ))
2351                    }
2352                }
2353            }) {
2354            Ok(coordinator) => coordinator,
2355            Err(error) => {
2356                context.stop_and_reject_pending();
2357                let worker = tokio::task::spawn_blocking(move || worker.join());
2358                let startup = ProxyError::Io(error);
2359                if let Err(cleanup_error) = join_native_thread("preconnect worker", worker).await {
2360                    return Err(ProxyError::Control(format!(
2361                        "{startup}; worker cleanup after coordinator startup failure failed: {cleanup_error}"
2362                    )));
2363                }
2364                return Err(startup);
2365            }
2366        };
2367        let coordinator = tokio::task::spawn_blocking(move || coordinator.join());
2368        let worker = tokio::task::spawn_blocking(move || worker.join());
2369        let mut sessions = JoinSet::new();
2370        loop {
2371            tokio::select! {
2372                _ = shutdown.changed() => break,
2373                handoff = handoff_receiver.recv() => {
2374                    let Some(handoff) = handoff else {
2375                        break;
2376                    };
2377                    match handoff.into_tokio() {
2378                        Ok((accepted, outbound, tuple, destination)) => {
2379                            sessions.spawn(async move {
2380                                if let Err(error) = bridge_tcp_streams(
2381                                    accepted,
2382                                    outbound,
2383                                    tuple,
2384                                    destination,
2385                                ).await {
2386                                    tracing::warn!(%error, "conditional TCP forwarding session failed");
2387                                }
2388                            });
2389                        }
2390                        Err(error) => tracing::warn!(%error, "conditional TCP socket handoff failed"),
2391                    }
2392                }
2393                Some(_) = sessions.join_next(), if !sessions.is_empty() => {}
2394            }
2395        }
2396        context.stop_and_reject_pending();
2397        // Closing the receiver drops every unconverted handoff and its
2398        // reservation before waiting for the native workers to finish.
2399        drop(handoff_receiver);
2400        sessions.abort_all();
2401        while sessions.join_next().await.is_some() {}
2402        let coordinator_result = join_native_thread("coordinator", coordinator).await;
2403        let worker_result = join_native_thread("preconnect worker", worker).await;
2404        coordinator_result?;
2405        worker_result
2406    }

2434    /// The condition callback only performs address decoding, bounded atomic
2435    /// admission, a nonblocking queue operation, and event signalling. It
2436    /// deliberately performs no Winsock, RPC, blocking, or re-entrant work.
2437    unsafe extern "system" fn condition_callback(
2438        caller_id: *mut WsaBuf,
2439        _caller_data: *mut WsaBuf,
2440        _caller_qos: *mut c_void,
2441        _callee_qos: *mut c_void,
2442        callee_id: *mut WsaBuf,
2443        _callee_data: *mut WsaBuf,
2444        _group: *mut u32,
2445        callback_data: usize,
2446    ) -> c_int {
2447        // SAFETY: WSAAccept supplies the same pointer passed as callback data;
2448        // the coordinator retains an Arc for all possible callback invocations.
2449        let context = unsafe { &*(callback_data as *const ConditionalContext) };
2450        let Some(source) = (unsafe { socket_addr_from_wsa_buf(caller_id) }) else {
2451            return CF_REJECT;
2452        };
2453        let Some(destination) = (unsafe { socket_addr_from_wsa_buf(callee_id) }) else {
2454            return CF_REJECT;
2455        };
2456        if source.is_ipv4() != destination.is_ipv4() {
2457            return CF_REJECT;
2458        }
2459        let tuple = Tuple {
2460            source,
2461            destination,
2462            protocol: TCP_PROTOCOL,
2463        };
2464        // `try_lock` is essential here: a contended coordinator table is a
2465        // resource failure, not a reason for callback blocking.
2466        let Ok(mut attempt) = context.attempt.try_lock() else {
2467            return CF_REJECT;
2468        };
2469        if let Some(slot) = attempt.as_ref() {
2470            if slot.attempt.id.tuple != tuple {
2471                return CF_REJECT;
2472            }
2473            if !context.accepting.load(Ordering::Acquire) {
2474                slot.attempt
2475                    .state
2476                    .store(ConditionalAttemptState::Cancelled as u8, Ordering::Release);
2477                return CF_REJECT;
2478            }
2479            match slot.attempt.state() {
2480                ConditionalAttemptState::Deferred => return CF_DEFER,
2481                ConditionalAttemptState::Ready => {
2482                    if !context.reserve_handoff() {
2483                        slot.attempt
2484                            .state
2485                            .store(ConditionalAttemptState::Rejected as u8, Ordering::Release);
2486                        return CF_REJECT;
2487                    }
2488                    if claim_ready_attempt(&slot.attempt) {
2489                        context
2490                            .accepted_generation
2491                            .store(slot.attempt.id.generation, Ordering::Release);
2492                        return CF_ACCEPT;
2493                    }
2494                    context.release_handoff_reservation();
2495                    slot.attempt
2496                        .state
2497                        .store(ConditionalAttemptState::Rejected as u8, Ordering::Release);
2498                    return CF_REJECT;
2499                }
2500                ConditionalAttemptState::Claimed => return CF_REJECT,
2501                ConditionalAttemptState::Rejected | ConditionalAttemptState::Cancelled => {
2502                    return CF_REJECT
2503                }
2504            }
2505        }
2506        if !context.accepting.load(Ordering::Acquire) {
2507            return CF_REJECT;
2508        }
2509        let Some(reservation) = context.admissions.reserve(tuple) else {
2510            return CF_REJECT;
2511        };
2512        // `try_send` is bounded and nonblocking. The fixed slot is installed
2513        // before the worker can observe the queue, so the callback performs
2514        // no heap allocation.
2515        *attempt = Some(AttemptSlot::new(reservation));
2516        if context.request_sender.try_send(()).is_err() {
2517            let removed = attempt.take().expect("installed conditional attempt");
2518            drop(attempt);
2519            context.reject_and_release(removed, ConditionalAttemptState::Rejected);
2520            return CF_REJECT;
2521        }
2522        drop(attempt);
2523        context.wake();
2524        CF_DEFER
2525    }

2527    /// Decodes the callback's sockaddr bytes without invoking Winsock.
2528    unsafe fn socket_addr_from_wsa_buf(buffer: *mut WsaBuf) -> Option<SocketAddr> {
2529        if buffer.is_null() || unsafe { (*buffer).buf.is_null() } {
2530            return None;
2531        }
2532        let buffer = unsafe { &*buffer };
2533        let bytes =
2534            unsafe { std::slice::from_raw_parts(buffer.buf.cast::<u8>(), buffer.len as usize) };
2535        if bytes.len() < 2 {
2536            return None;
2537        }
2538        let family = u16::from_ne_bytes([bytes[0], bytes[1]]);
2539        let port = bytes
2540            .get(2..4)
2541            .map(|bytes| u16::from_be_bytes([bytes[0], bytes[1]]))?;
2542        match family {
2543            2 if bytes.len() >= 16 => Some(SocketAddr::from((
2544                std::net::Ipv4Addr::new(bytes[4], bytes[5], bytes[6], bytes[7]),
2545                port,
2546            ))),
2547            23 if bytes.len() >= 28 => {
2548                let address = std::net::Ipv6Addr::from(<[u8; 16]>::try_from(&bytes[8..24]).ok()?);
2549                let scope_id = u32::from_ne_bytes(bytes[24..28].try_into().ok()?);
2550                Some(SocketAddr::V6(SocketAddrV6::new(
2551                    address, port, 0, scope_id,
2552                )))
2553            }
2554            _ => None,
2555        }
2556    }
