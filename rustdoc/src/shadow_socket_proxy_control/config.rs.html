17#[derive(Debug, Clone, PartialEq, Eq)]
18/// Listener address plus the wildcard flags required by the BPF ABI.
19pub struct ListenerDescriptor {
20    /// Address on which the control transport listens.
21    pub address: IpAddr,
22    /// Non-zero control listener port.
23    pub port: u16,
24    /// True only when `address` is an IPv4 unspecified address.
25    pub ipv4_wildcard: bool,
26    /// True only when `address` is an IPv6 unspecified address.
27    pub ipv6_wildcard: bool,
28}
29
30impl ListenerDescriptor {
31    /// Splits a socket address and derives family-specific wildcard flags.
32    pub fn from_socket_addr(address: SocketAddr) -> Self {
33        Self {
34            address: address.ip(),
35            port: address.port(),
36            ipv4_wildcard: address.ip().is_unspecified() && address.is_ipv4(),
37            ipv6_wildcard: address.ip().is_unspecified() && address.is_ipv6(),
38        }
39    }
40
41    /// Reconstructs the socket address represented by this descriptor.
42    pub fn socket_addr(&self) -> SocketAddr {
43        SocketAddr::new(self.address, self.port)
44    }
45}

47#[derive(Debug, Clone, PartialEq, Eq)]
48/// Validated settings written to the BPF runtime-config map and read by workers.
49pub struct RuntimeConfig {
50    /// Runtime-config ABI version required by the BPF program.
51    pub schema_version: u16,
52    /// Monotonic revision assigned when the store publishes a configuration.
53    pub revision: u64,
54    /// Delay between maintenance scans.
55    pub cleanup_interval: Duration,
56    /// Age after which an otherwise active flow or mapping is eligible for cleanup.
57    pub idle_ttl: Duration,
58    /// Maximum number of map records examined by one cleanup pass.
59    pub map_scan_batch: usize,
60    /// Maximum number of log records retained by the bounded ring.
61    pub log_capacity: usize,
62    /// Requested active-flow capacity, bounded by the loaded map maxima.
63    pub active_flow_capacity: usize,
64    /// Grace period retained after both TCP FIN exchanges complete.
65    pub tcp_terminal_grace: Duration,
66    /// Optional IPv4 redirection target; address and port must be paired.
67    pub ipv4_target: Option<SocketAddr>,
68    /// Optional IPv6 redirection target; address and port must be paired.
69    pub ipv6_target: Option<SocketAddr>,
70    /// Immutable control listener descriptor.
71    pub listener: ListenerDescriptor,
72}
73
74impl Default for RuntimeConfig {
75    /// Returns the protocol-compatible baseline configuration for port 50051.
76    fn default() -> Self {
77        Self {
78            schema_version: RUNTIME_CONFIG_ABI_VERSION,
79            revision: 1,
80            cleanup_interval: Duration::from_secs(10),
81            idle_ttl: Duration::from_secs(60),
82            map_scan_batch: 256,
83            log_capacity: 1024,
84            active_flow_capacity: 4096,
85            tcp_terminal_grace: Duration::from_secs(30),
86            ipv4_target: None,
87            ipv6_target: None,
88            listener: ListenerDescriptor::from_socket_addr(
89                "0.0.0.0:50051".parse().expect("valid default listener"),
90            ),
91        }
92    }
93}

95#[derive(Debug, Error, Clone, PartialEq, Eq)]
96/// Validation failures that prevent an unsafe or ABI-incompatible configuration.
97pub enum ConfigError {
98    #[error("cleanup interval must be non-zero")]
99    /// Maintenance cannot run with a zero interval.
100    ZeroCleanupInterval,
101    #[error("runtime config schema version is unsupported")]
102    /// The requested runtime-config ABI is not understood.
103    UnsupportedSchemaVersion,
104    #[error("idle TTL must be non-zero")]
105    /// Flow cleanup requires a non-zero idle lifetime.
106    ZeroIdleTtl,
107    #[error("map scan batch must be between 1 and 100000")]
108    /// The scan batch is outside the supported 1..=100000 range.
109    InvalidBatch,
110    #[error("log capacity must be between 1 and 100000")]
111    /// The log ring capacity is outside the supported 1..=100000 range.
112    InvalidLogCapacity,
113    #[error("cleanup interval must not exceed idle TTL")]
114    /// A cleanup interval longer than the idle TTL could miss expiration.
115    IntervalExceedsTtl,
116    #[error("active flow capacity must be between 1 and the ELF maximum")]
117    /// Active-flow capacity is zero or exceeds the state-map maximum.
118    InvalidFlowCapacity,
119    #[error("active flow capacity requires three flow indexes per flow")]
120    /// Three tuple indexes per active flow would exceed the index-map maximum.
121    FlowIndexCapacityExceeded,
122    #[error("TCP terminal grace must be non-zero")]
123    /// TCP terminal retention requires a non-zero grace period.
124    ZeroTerminalGrace,
125    #[error("runtime duration is too large")]
126    /// A duration cannot be represented safely by the BPF nanosecond fields.
127    DurationOverflow,
128    #[error("listener port must be non-zero")]
129    /// The control listener port is zero.
130    InvalidListenerPort,
131    #[error("listener wildcard flags do not match the listener address")]
132    /// Wildcard flags disagree with the listener address family or value.
133    InvalidListenerWildcard,
134    #[error("target address and port must be set together")]
135    /// A target supplied only an address or only a port.
136    PartialTarget,
137    #[error("target address family does not match its field")]
138    /// An IPv4/IPv6 target was supplied in the opposite target field.
139    TargetFamilyMismatch,
140    #[error("target address is unspecified")]
141    /// A target address is unspecified and cannot identify a destination.
142    UnspecifiedTarget,
143}

145impl RuntimeConfig {
146    /// Validates against the compiled-in map capacities.
147    pub fn validate(&self) -> Result<(), ConfigError> {
148        self.validate_with_maxima(MapMaxima::default())
149    }
150
151    /// Validates ABI versions, bounds, address families, and map capacity
152    /// constraints before publication or BPF writes.
153    pub fn validate_with_maxima(&self, maxima: MapMaxima) -> Result<(), ConfigError> {
154        if self.schema_version != RUNTIME_CONFIG_ABI_VERSION {
155            return Err(ConfigError::UnsupportedSchemaVersion);
156        }
157        if self.cleanup_interval.is_zero() {
158            return Err(ConfigError::ZeroCleanupInterval);
159        }
160        if self.idle_ttl.is_zero() {
161            return Err(ConfigError::ZeroIdleTtl);
162        }
163        if self.map_scan_batch == 0 || self.map_scan_batch > 100_000 {
164            return Err(ConfigError::InvalidBatch);
165        }
166        if self.log_capacity == 0 || self.log_capacity > 100_000 {
167            return Err(ConfigError::InvalidLogCapacity);
168        }
169        if self.cleanup_interval > self.idle_ttl {
170            return Err(ConfigError::IntervalExceedsTtl);
171        }
172        if self.active_flow_capacity == 0 || self.active_flow_capacity > maxima.flow_state {
173            return Err(ConfigError::InvalidFlowCapacity);
174        }
175        if self.active_flow_capacity.saturating_mul(3) > maxima.flow_index {
176            return Err(ConfigError::FlowIndexCapacityExceeded);
177        }
178        if self.tcp_terminal_grace.is_zero() {
179            return Err(ConfigError::ZeroTerminalGrace);
180        }
181        if self.listener.port == 0 {
182            return Err(ConfigError::InvalidListenerPort);
183        }
184        if self.listener.ipv4_wildcard
185            != (self.listener.address.is_unspecified() && self.listener.address.is_ipv4())
186            || self.listener.ipv6_wildcard
187                != (self.listener.address.is_unspecified() && self.listener.address.is_ipv6())
188        {
189            return Err(ConfigError::InvalidListenerWildcard);
190        }
191        validate_target(self.ipv4_target, true)?;
192        validate_target(self.ipv6_target, false)?;
193        if self.cleanup_interval.as_secs() > 365 * 24 * 60 * 60
194            || self.idle_ttl.as_secs() > 365 * 24 * 60 * 60
195            || self.tcp_terminal_grace.as_secs() > 365 * 24 * 60 * 60
196        {
197            return Err(ConfigError::DurationOverflow);
198        }
199        Ok(())
200    }
201}
202
203/// Checks optional target pairing, family, port, and address specificity.
204fn validate_target(target: Option<SocketAddr>, ipv4: bool) -> Result<(), ConfigError> {
205    let Some(target) = target else {
206        return Ok(());
207    };
208    if target.port() == 0 {
209        return Err(ConfigError::PartialTarget);
210    }
211    if target.ip().is_ipv4() != ipv4 {
212        return Err(ConfigError::TargetFamilyMismatch);
213    }
214    if target.ip().is_unspecified() {
215        return Err(ConfigError::UnspecifiedTarget);
216    }
217    Ok(())
218}

220/// Atomically published configuration with revision assignment and map maxima.
221pub struct ConfigStore {
222    /// Last validated configuration visible to readers.
223    current: ArcSwap<RuntimeConfig>,
224    /// Counter used to assign the next published revision.
225    next_revision: AtomicU64,
226    /// Map capacities used to validate every update.
227    maxima: MapMaxima,
228}
229
230impl ConfigStore {
231    /// Creates a store using the shipped BPF map capacities.
232    pub fn new(initial: RuntimeConfig) -> Result<Self, ConfigError> {
233        Self::new_with_maxima(initial, MapMaxima::default())
234    }
235
236    /// Validates and publishes an initial configuration against explicit maxima.
237    pub fn new_with_maxima(initial: RuntimeConfig, maxima: MapMaxima) -> Result<Self, ConfigError> {
238        initial.validate_with_maxima(maxima)?;
239        Ok(Self {
240            next_revision: AtomicU64::new(initial.revision + 1),
241            current: ArcSwap::from_pointee(initial),
242            maxima,
243        })
244    }
245
246    /// Loads a consistent, reference-counted configuration snapshot.
247    pub fn snapshot(&self) -> std::sync::Arc<RuntimeConfig> {
248        self.current.load_full()
249    }
250
251    /// Validates, assigns a fresh revision, and atomically publishes `next`.
252    pub fn update(
253        &self,
254        mut next: RuntimeConfig,
255    ) -> Result<std::sync::Arc<RuntimeConfig>, ConfigError> {
256        next.validate_with_maxima(self.maxima)?;
257        next.revision = self.next_revision.fetch_add(1, Ordering::SeqCst);
258        let next = std::sync::Arc::new(next);
259        self.current.store(next.clone());
260        Ok(next)
261    }
262
263    /// Returns the map capacities enforced by this store.
264    pub fn maxima(&self) -> MapMaxima {
265        self.maxima
266    }
267
268    /// Revalidates the current snapshot against capacities reported after attach.
269    pub fn validate_snapshot_with_maxima(&self, maxima: MapMaxima) -> Result<(), ConfigError> {
270        self.snapshot().validate_with_maxima(maxima)
271    }
272}

274#[cfg(test)]
275mod tests {
276    use super::*;
277
278    #[test]
279    fn updates_atomically_and_rejects_invalid_values() {
280        let store = ConfigStore::new(RuntimeConfig::default()).unwrap();
281        let mut next = (*store.snapshot()).clone();
282        next.idle_ttl = Duration::from_secs(120);
283        let updated = store.update(next).unwrap();
284        assert_eq!(updated.revision, 2);
285
286        let mut invalid = (*updated).clone();
287        invalid.map_scan_batch = 0;
288        assert_eq!(store.update(invalid), Err(ConfigError::InvalidBatch));
289        assert_eq!(store.snapshot().revision, 2);
290    }
291
292    #[test]
293    fn runtime_caps_cannot_exceed_fixed_elf_maxima() {
294        let config = RuntimeConfig {
295            active_flow_capacity: MapMaxima::default().flow_state + 1,
296            ..RuntimeConfig::default()
297        };
298        assert!(matches!(
299            ConfigStore::new(config),
300            Err(ConfigError::InvalidFlowCapacity)
301        ));
302    }
303
304    #[test]
305    fn target_pairs_are_atomic() {
306        let config = RuntimeConfig {
307            ipv4_target: Some("192.0.2.10:0".parse().unwrap()),
308            ..RuntimeConfig::default()
309        };
310        assert!(matches!(
311            ConfigStore::new(config),
312            Err(ConfigError::PartialTarget)
313        ));
314    }

316    #[test]
317    fn listener_descriptor_round_trips_to_socket_address() {
318        let address = "192.0.2.10:50051".parse().unwrap();
319        let descriptor = ListenerDescriptor::from_socket_addr(address);
320        assert_eq!(descriptor.socket_addr(), address);
